# Get all books
curl -X GET http://localhost:3000/api/books

# Get a specific book
curl -X GET http://localhost:3000/api/books/1

# Add a new book
curl -X POST http://localhost:3000/api/books \
-H "Content-Type: application/json" \
-d '{"title": "Sample Book", "author": "John Doe", "isbn": "1234567890", "publication_year": 2023, "genre": "Fiction"}'

# Update a book
curl -X PUT http://localhost:3000/api/books/1 \
-H "Content-Type: application/json" \
-d '{"title": "Updated Title", "author": "Jane Doe", "isbn": "1234567890", "publication_year": 2022, "genre": "Non-Fiction"}'

# Delete a book
curl -X DELETE http://localhost:3000/api/books/1
